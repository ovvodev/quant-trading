// Safe division — prevent na and division by zero
safeDiv(float num, float den, float fallback = 0.0) =>
    den != 0 and not na(num) and not na(den) ? num / den : fallback

// Ehlers Super Smoother (2-pole)
ehlersSS(float src, int period) =>
    float a1 = math.exp(-math.sqrt(2) * math.pi / math.max(period, 2))
    float b1 = 2.0 * a1 * math.cos(math.sqrt(2) * math.pi / math.max(period, 2))
    float c2 = b1
    float c3 = -a1 * a1
    float c1 = 1.0 - c2 - c3
    var float ss = na
    ss := bar_index < 3 ? nz(src, 0.0) : c1 * (src + nz(src[1], src)) * 0.5 + c2 * nz(ss[1], src) + c3 * nz(ss[2], src)
    ss

// Adaptive ATR using Kaufman efficiency ratio
adaptiveAtr(int period) =>
    float rawAtr = nz(ta.atr(period), 0.0)
    float direction = math.abs(close - close[period])
    float volatility = math.sum(math.abs(close - close[1]), period)
    float er = safeDiv(direction, volatility, 0.5)
    float fastSc = 2.0 / 3.0
    float slowSc = 2.0 / 31.0
    float sc = math.pow(er * (fastSc - slowSc) + slowSc, 2)
    var float aAtr = na
    aAtr := na(aAtr[1]) ? rawAtr : aAtr[1] + sc * (rawAtr - aAtr[1])
    aAtr

// Adaptive Volatility Trend Engine (AVTE)
avte(float src, float factor, int atrLen) =>
    float filteredPrice = ehlersSS(src, math.max(math.round(atrLen / 2), 3))
    float aAtr = adaptiveAtr(atrLen)
    float upperBand = filteredPrice + factor * aAtr
    float lowerBand = filteredPrice - factor * aAtr

    var float upper = na
    var float lower = na
    var int dir = 1
    var float trendLine = na

    lower := nz(lower[1], lowerBand)
    upper := nz(upper[1], upperBand)

    lower := lowerBand > lower or close[1] < lower ? lowerBand : lower
    upper := upperBand < upper or close[1] > upper ? upperBand : upper

    if close > upper
        dir := 1
    else if close < lower
        dir := -1

    trendLine := dir == 1 ? lower : upper

    [trendLine, dir, filteredPrice, aAtr]

// ══════════════════════════════════════
// MOMENTUM SCORE — Symmetric (0-100)
// ══════════════════════════════════════
calcMomentumScore() =>
    float score = 50.0

    // Factor 1: EMA alignment (±15)
    bool emaAlignBull = close > ema50 and ema50 > ema200
    bool emaAlignBear = close < ema50 and ema50 < ema200

    score += emaAlignBull ? 15.0 : emaAlignBear ? -15.0 : 0.0

    // Factor 2: RSI position (±10)
    score += (rsi14 > 50 and rsi14 < 70) ? 10.0 :
             (rsi14 <= 50 and rsi14 > 30) ? -10.0 : 0.0

    // Factor 3: MACD histogram (±12)
    score += (macdHist > 0 and macdHist > macdHist[1]) ? 12.0 :
             (macdHist < 0 and macdHist < macdHist[1]) ? -12.0 : 0.0

    // Factor 4: Volume confirmation (±8)
    bool volAbove = hasVolume and volume > volSma20 * 1.2
    bool volBelow = hasVolume and volume < volSma20 * 0.8

    float priceChg = close - close[1]

    score += volAbove and priceChg > 0 ? 8.0 :
             volAbove and priceChg < 0 ? -8.0 :
             volBelow ? (priceChg > 0 ? -4.0 : 4.0) : 0.0

    // Factor 5: ADX strength (±8)
    bool adxTrending = adxSafe > 20

    score += adxTrending and diP > diM ? 8.0 :
             adxTrending and diM > diP ? -8.0 : 0.0

    math.max(0.0, math.min(100.0, score))
