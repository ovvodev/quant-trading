rangesOverlap(top1, btm1, top2, btm2) =>
    math.max(btm1, btm2) < math.min(top1, top2)

type ExecHtfFvg
    float top
    float btm
    bool  isBull

var array<ExecHtfFvg> execHtfFvgs = array.new<ExecHtfFvg>()

f_execHtfFvg() =>
    bool ib = low[1] > high[3]
    bool ir = high[1] < low[3]
    [ib, low[1], high[3], ir, low[3], high[1], time[3]]

f_trackExecHtf(string tf) =>
    [ib, bTop, bBtm, ir, sTop, sBtm, gapTime] = request.security(syminfo.tickerid, tf, f_execHtfFvg(), lookahead=barmerge.lookahead_on)
    if gapTime != gapTime[1]
        if ib
            array.push(execHtfFvgs, ExecHtfFvg.new(bTop, bBtm, true))
        if ir
            array.push(execHtfFvgs, ExecHtfFvg.new(sTop, sBtm, false))

f_trackExecHtf("15")
f_trackExecHtf("60")

if array.size(execHtfFvgs) > 0
    for i = array.size(execHtfFvgs) - 1 to 0
        ExecHtfFvg f = array.get(execHtfFvgs, i)
        if (f.isBull and low <= f.btm) or (not f.isBull and high >= f.top)
            array.remove(execHtfFvgs, i)
while array.size(execHtfFvgs) > 40
    array.shift(execHtfFvgs)

f_execTargetAbove(float px) =>
    float t = na
    if array.size(execHtfFvgs) > 0
        for i = 0 to array.size(execHtfFvgs) - 1
            ExecHtfFvg f = array.get(execHtfFvgs, i)
            if f.btm > px and (na(t) or f.btm < t)
                t := f.btm
    t

f_execTargetBelow(float px) =>
    float t = na
    if array.size(execHtfFvgs) > 0
        for i = 0 to array.size(execHtfFvgs) - 1
            ExecHtfFvg f = array.get(execHtfFvgs, i)
            if f.top < px and (na(t) or f.top > t)
                t := f.top
    t

// --- HTF Bias (moved before Section 3 so confluence score can use it) ---
f_calc_bias() =>
    var array<float> gTop    = array.new<float>()
    var array<float> gBtm    = array.new<float>()
    var array<int>   gDir    = array.new<int>()
    var array<bool>  gTested = array.new<bool>()

    if low[1] > high[3]
        array.push(gTop, low[1]),  array.push(gBtm, high[3])
        array.push(gDir, 1),       array.push(gTested, false)
    if high[1] < low[3]
        array.push(gTop, low[3]),  array.push(gBtm, high[1])
        array.push(gDir, -1),      array.push(gTested, false)

    while array.size(gDir) > 40
        array.shift(gTop), array.shift(gBtm)
        array.shift(gDir), array.shift(gTested)

    if array.size(gDir) > 0
        for i = array.size(gDir) - 1 to 0
            float gt = array.get(gTop, i)
            float gb = array.get(gBtm, i)
            int   gd = array.get(gDir, i)
            bool dead = gd == 1 ? close[1] < gb : close[1] > gt
            if dead
                array.remove(gTop, i), array.remove(gBtm, i)
                array.remove(gDir, i), array.remove(gTested, i)
            else
                if gd == 1 ? low[1] < gt : high[1] > gb
                    array.set(gTested, i, true)

    int bias = 0
    if array.size(gDir) > 0
        for i = array.size(gDir) - 1 to 0
            if array.get(gTested, i)
                bias := array.get(gDir, i)
                break
    bias

int bias5m  = request.security(syminfo.tickerid, "5",   f_calc_bias(), lookahead=barmerge.lookahead_on)

// =============================================================================
// 3. IFVG CORE LOGIC (STRICT: CANDLE-3 + OVERLAP + SWEEP)
// =============================================================================

isBullFVG = low  > high[2]
isBearFVG = high < low[2]

// FIX #5: flag pattern — alert called once AFTER the loop
if barstate.isconfirmed and (isBullFVG or isBearFVG)
    newTop = isBullFVG ? low   : low[2]
    newBtm = isBullFVG ? high[2] : high

    bool  af_bull          = false
    bool  af_bear          = false
    float af_invGap        = na
    bool  af_isAplus       = false
    float af_c1High        = na
    float af_c1Low         = na
    float af_initialSL     = na
    float af_swingSL       = na
    float af_aplusTarget   = na
    float af_sweepExtreme  = na

    for i = 1 to setupLookback
        bool oldIsBull = low[i]  > high[i+2]
        bool oldIsBear = high[i] < low[i+2]

        if (isBullFVG and oldIsBear) or (isBearFVG and oldIsBull)
            oldTop = oldIsBull ? low[i]    : low[i+2]
            oldBtm = oldIsBull ? high[i+2] : high[i]

            if rangesOverlap(newTop, newBtm, oldTop, oldBtm)
                bool  sweepFound   = false
                float sweepExtreme = na

                if isBullFVG
                    float lowestInSeq = low
                    for k = 0 to i
                        lowestInSeq := math.min(lowestInSeq, low[k])
                    float preExistingLow = low[i+1]
                    for k = 1 to liqLookback
                        preExistingLow := math.min(preExistingLow, low[i+k])
                    if lowestInSeq < preExistingLow
                        sweepFound := true
                    sweepExtreme := lowestInSeq
                else
                    float highestInSeq = high
                    for k = 0 to i
                        highestInSeq := math.max(highestInSeq, high[k])
                    float preExistingHigh = high[i+1]
                    for k = 1 to liqLookback
                        preExistingHigh := math.max(preExistingHigh, high[i+k])
                    if highestInSeq > preExistingHigh
                        sweepFound := true
                    sweepExtreme := highestInSeq

                if sweepFound
                    bool isAplus = isBullFVG
                         ? (not na(lastBullSweepBar) and (bar_index - lastBullSweepBar) <= setupLookback)
                         : (not na(lastBearSweepBar) and (bar_index - lastBearSweepBar) <= setupLookback)
                    bool aplus_ok = not showAplus or isAplus

                    if aplus_ok
                        color  boxColor  = isBullFVG ? bullColor : bearColor
                        string txtSymbol = (isAplus ? "A+ " : "") + (isBullFVG ? "+IFVG" : "-IFVG")
                        float  visTop    = math.min(oldTop, newTop)
                        float  visBtm   = math.max(oldBtm, newBtm)
                        float  midPoint  = (visTop + visBtm) / 2

                        box b = box.new(left=bar_index[i+2], top=visTop, bottom=visBtm, right=bar_index + lookForward, border_color=boxColor, bgcolor=boxColor)
                        label l = label.new(bar_index + lookForward, midPoint, txtSymbol, color=color.new(color.white, 100), textcolor=txtColor, style=label.style_none, size=lblSize)

                        line ceLine = na
                        if showCE
                            ceLine := line.new(bar_index[i+2], midPoint, bar_index + lookForward, midPoint, color=color.new(txtColor, 50), style=line.style_dashed)

                        array.push(activeSetups, IfvgSetup.new(b, l, ceLine, isBullFVG, visTop, visBtm, true, false, false))

                        float invGap = oldTop - oldBtm
                        if invGap >= minImbalanceTicks * syminfo.mintick
                            if isBullFVG
                                af_bull := true
                            else
                                af_bear := true
                            af_invGap      := invGap
                            af_isAplus     := isAplus
                            af_c1High      := high[2]
                            af_c1Low       := low[2]
                            af_initialSL   := isBullFVG ? low[2]  : high[2]
                            af_swingSL     := isBullFVG ? swingSLLong : swingSLShort
                            af_sweepExtreme := sweepExtreme
                            af_aplusTarget := isAplus ? (isBullFVG ? f_execTargetAbove(close) : f_execTargetBelow(close)) : na

                    break

    // Fire alerts once after the loop
    if af_bull or af_bear
        string actionTxt = af_bull ? "buy"    : "sell"
        string ifvgTyp   = af_bull ? "Bullish" : "Bearish"
        int    sigDir    = af_bull ? 1 : -1

        // [S5] Confluence score 0-5
        bool inAnySession = not na(time(timeframe.period, sessAsiaIn, sessTz)) or not na(time(timeframe.period, sessLondonIn, sessTz)) or not na(time(timeframe.period, sessNYIn, sessTz))
        int confluence = 0
        if af_isAplus
            confluence += 1
        if inAnySession
            confluence += 1
        if minImbalanceTicks > 0 and af_invGap >= 2 * minImbalanceTicks * syminfo.mintick
            confluence += 1
        if bias1h == sigDir
            confluence += 1
        if bias4h == sigDir
            confluence += 1

        // [S6] R:R ratio
        float rrRatio = na
        if af_isAplus and not na(af_aplusTarget) and not na(af_initialSL)
            float risk   = math.abs(close - af_initialSL)
            float reward = math.abs(af_aplusTarget - close)
            if risk > 0
                rrRatio := reward / risk

        if wantText
            string defaultTxt = ifvgTyp + " IFVG Detected!"
            string customTxt  = af_bull ? msgBull : msgBear
            alert(str.length(customTxt) > 0 ? customTxt : defaultTxt, alert.freq_once_per_bar_close)

        if wantJson
            string rrStr  = not na(rrRatio) ? str.tostring(rrRatio, "#.##") : "0"
            string tgtStr = af_isAplus and not na(af_aplusTarget) ? str.tostring(af_aplusTarget, "#.00") : "0"
            alert('{"secret":"'         + webhookSecret
                 + '","symbol":"'       + symTicker
                 + '","action":"'       + actionTxt
                 + '","timeframe":"'    + tfString
                 + '","ifvg_type":"'    + ifvgTyp
                 + '","sl_price":'      + str.tostring(af_initialSL,    "#.00")
                 + ',"c1_high":'        + str.tostring(af_c1High,       "#.00")
                 + ',"c1_low":'         + str.tostring(af_c1Low,        "#.00")
                 + ',"swing_sl":'       + str.tostring(af_swingSL,      "#.00")
                 + ',"entry_ref":'      + str.tostring(close,           "#.00")
                 + ',"imbalance":'      + str.tostring(af_invGap / syminfo.mintick, "#.##")
                 + ',"sweep_extreme":'  + str.tostring(af_sweepExtreme, "#.00")
                 + ',"a_plus":'         + (af_isAplus ? "true" : "false")
                 + ',"a_plus_target":'  + tgtStr
                 + ',"confluence":'     + str.tostring(confluence)
                 + ',"rr":'             + rrStr
                 + '}', alert.freq_once_per_bar_close)

        // Register stop in the array (FIX #3)
        array.push(execInvs, ExecInv.new(af_initialSL, sigDir))
        while array.size(execInvs) > 10
            array.shift(execInvs)

        if af_isAplus and showStopPlacement
            line.new(bar_index, af_initialSL, bar_index + lookForward, af_initialSL, color=color.new(color.orange, 0), width=2)
            label.new(bar_index + lookForward, af_initialSL, "A+ C1 SL", style=label.style_label_left, textcolor=color.white, color=color.new(color.orange, 20), size=size.tiny)
            line.new(bar_index, af_swingSL, bar_index + lookForward, af_swingSL, color=color.new(color.red, 0), width=2, style=line.style_dashed)
            label.new(bar_index + lookForward, af_swingSL, "A+ swing", style=label.style_label_left, textcolor=color.white, color=color.new(color.red, 20), size=size.tiny)

// FIX #3: Iterate ALL active ExecInv records for invalidation
if barstate.isconfirmed and array.size(execInvs) > 0
    for i = array.size(execInvs) - 1 to 0
        ExecInv inv = array.get(execInvs, i)
        bool triggered = (inv.dir == 1 and close < inv.level) or (inv.dir == -1 and close > inv.level)
        if triggered
            if wantJson
                string invDir = inv.dir == 1 ? "bullish" : "bearish"
                alert('{"secret":"'   + webhookSecret
                     + '","symbol":"' + symTicker
                     + '","action":"exit","timeframe":"' + tfString
                     + '","reason":"' + invDir + ' IFVG invalidated after candle-3 confirmation"}'
                     , alert.freq_once_per_bar_close)
            if showInvalidation
                line.new(bar_index, close, bar_index + 3, close, color=txtColor, width=2)
                label.new(bar_index + 3, close, "invalidated", style=label.style_none, textcolor=txtColor, size=size.tiny)
            array.remove(execInvs, i)

// =============================================================================
// 4. INVALIDATION MANAGEMENT + [S1] RETEST + [S2] CE TOUCH ALERTS
// =============================================================================

if array.size(activeSetups) > 0
    for i = array.size(activeSetups) - 1 to 0
        s = array.get(activeSetups, i)

        if s.isValid and barstate.isconfirmed
            bool failed = s.isBullish ? close < s.realBtm : close > s.realTop

            if failed
                if delInvalid
                    box.delete(s.id)
                    label.delete(s.lbl)
                    line.delete(s.ce)
                    array.remove(activeSetups, i)
                else
                    box.set_bgcolor(s.id, failColor)
                    box.set_border_color(s.id, failColor)
                    label.set_textcolor(s.lbl, color.new(txtColor, 50))
                    line.delete(s.ce)
                    s.isValid := false
                    array.set(activeSetups, i, s)

            else
                // [S1] IFVG Retest — fires once when price re-enters the box
                if not s.retested
                    bool retestBull = s.isBullish      and low  <= s.realTop and close >= s.realBtm
                    bool retestBear = not s.isBullish  and high >= s.realBtm and close <= s.realTop
                    if retestBull or retestBear
                        s.retested := true
                        array.set(activeSetups, i, s)
                        if alertRetest
                            string rDir = s.isBullish ? "Bullish" : "Bearish"
                            if wantText
                                alert(rDir + " IFVG Retest — price re-entered the gap!", alert.freq_once_per_bar_close)
                            if wantJson
                                alert('{"secret":"'      + webhookSecret
                                     + '","symbol":"'    + symTicker
                                     + '","action":"retest","timeframe":"' + tfString

    outTxt

f_htfDraw(HtfFvg f) =>
    color  fillCol   = f.isBull ? (f.filled ? htfBullFilled : htfBullColor) : (f.filled ? htfBearFilled : htfBearColor)
    color  edgeCol   = f.filled ? (f.isBull ? htfBullEdge : htfBearEdge) : fillCol
    string edgeStyle = f.filled ? line.style_dotted : line.style_solid
    int    rightX    = f.filled ? f.fillTime : htfRightX
    string txt       = f.tfLabel + (f.isBull ? " Bullish FVG" : " Bearish FVG") + (f.filled ? " (Filled)" : "")
    color  txtCol    = f.filled ? color.new(txtColor, 40) : txtColor
    if na(f.id)
        f.id  := box.new(left=f.startTime, top=f.top, bottom=f.btm, right=rightX, xloc=xloc.bar_time, border_color=edgeCol, bgcolor=fillCol, border_style=edgeStyle)

    else
        box.set_bgcolor(f.id, fillCol), box.set_border_color(f.id, edgeCol), box.set_border_style(f.id, edgeStyle), box.set_right(f.id, rightX)
        label.set_x(f.lbl, rightX), label.set_text(f.lbl, txt), label.set_textcolor(f.lbl, txtCol)
    true
